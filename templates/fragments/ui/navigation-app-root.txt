package {{packageName}}.ui.navigation

import androidx.compose.runtime.Composable
import androidx.navigation3.runtime.NavKey
import androidx.navigation3.runtime.entryProvider
import androidx.navigation3.runtime.rememberNavBackStack
import androidx.navigation3.ui.NavDisplay
import kotlinx.serialization.Serializable

// @initializr-slot app-imports

@Serializable
sealed interface AppRoute : NavKey {
    @Serializable
    data object Welcome : AppRoute

    // @initializr-slot app-routes
}

@Composable
fun AppRoot() {
    val backStack = rememberNavBackStack(AppRoute.{{initialRoute}})
    NavDisplay(
        backStack = backStack,
        onBack = { if (backStack.size > 1) backStack.removeLastOrNull() },
        entryProvider = entryProvider {
            entry<AppRoute.Welcome> { WelcomeScreen() }
            // @initializr-slot app-entries
        },
    )
}
