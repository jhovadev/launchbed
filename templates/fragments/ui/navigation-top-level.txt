fun shouldShowBottomBar(current: NavKey?): Boolean =
    TopLevelRoutes.size >= 2 && current is AppRoute && current in TopLevelRoutes

/**
 * Switches tab: the stack becomes [home] or [home, destination], so system Back from any tab
 * returns to the home tab before leaving the app.
 */
internal fun MutableList<NavKey>.selectTopLevel(destination: AppRoute, home: AppRoute = TopLevelRoutes.first()) {
    if (lastOrNull() == destination) return
    val target: List<NavKey> = if (destination == home) listOf(home) else listOf(home, destination)
    addAll(target)
    while (size > target.size) {
        removeAt(0)
    }
}

@Composable
fun AppBottomBar(current: NavKey?, onSelect: (AppRoute) -> Unit, modifier: Modifier = Modifier) {
    NavigationBar(modifier = modifier.testTag("bottom-navigation")) {
        TopLevelRoutes.forEach { route ->
            NavigationBarItem(
                selected = current == route,
                onClick = { onSelect(route) },
                icon = { route.topLevelIcon()?.let { Icon(imageVector = it, contentDescription = null) } },
                label = { Text(route.topLevelLabel()) },
            )
        }
    }
}
